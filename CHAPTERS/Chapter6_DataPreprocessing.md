as np\n\ndef apply_bandpass_filter(data, fs, lowcut=0.5, highcut=70, order=4):\n    \"\"\"\n    Apply bandpass filter to EEG data\n    \n    Args:\n        data: (n_channels, n_samples)\n        fs: Sampling frequency\n        lowcut: Low cutoff frequency (Hz)\n        highcut: High cutoff frequency (Hz)\n        order: Filter order\n    \n    Returns:\n        filtered_data: Bandpass filtered data\n    \"\"\"\n    # Design filter\n    nyquist = fs / 2\n    low = lowcut / nyquist\n    high = highcut / nyquist\n    \n    b, a = signal.butter(order, [low, high], btype='band')\n    \n    # Apply filter to each channel\n    filtered_data = np.zeros_like(data)\n    for ch in range(data.shape[0]):\n        filtered_data[ch] = signal.filtfilt(b, a, data[ch])\n    \n    return filtered_data\n\n# Usage\nfiltered_data = apply_bandpass_filter(data, fs=500, lowcut=0.5, highcut=70)\n```\n\n**Visualization:**\n\n```python\nimport matplotlib.pyplot as plt\n\n# Plot before/after Apply notch filter to remove power line noise\n    \n    Args:\n        data: (n_channels, n_samples)\n        fs: Sampling frequency\n        freq: Frequency to remove (50 or 60 Hz)\n        Q: Quality factor\n    \n    Returns:\n        filtered_data: Notch filtered data\n    \"\"\"\n    # Design notch filter\n    w0 = freq / (fs / 2)  # Normalized frequency\n    b, a = signal.iirnotch(w0, 6.5.1 Fixed Window Segmentation\n\n```python\ndef segment_eeg_fixed_window(data, fs, window_duration=4, overlap=0.5):\n    \"\"\"\n    Segment EEG into fixed-duration windows\n    \n    Args:\n        data: (n_channels, n_samples)\n        fs: Sampling frequency\n        window_duration: Window length in seconds\n        overlap: Overlap ratio (0-1)\n    \n    Returns:\n        segments: List of (n_channels, window_samples) arrays\n    \"\"\"\n    window_samples = int(window_duration * fs)\n    step_samples = int(window_samples * (1 - overlap))\n    \n    n_segments = (data.shape[1] - window_samples) // step_samples + 1\n    segments = []\n    \n    for i in range(n_segments):\n        start = i * step_samples\n        end = start + window_samples\n        if end <= data.shape[1]:\n            segments.append(data[:, start:end])\n    \n    return np.array(segments)\n\n# Usage: 4-second windows with 50% overlap\nsegments = segment_eeg_fixed_window(filtered_data, Segments\n\n```python\ndef remove_artifact_segments(segments, amplitude_threshold=200):\n    \"\"\"\n    Remove segments with artifacts\n    \n    Args:\n        segments: Array of (n_segments, of valid segments\n    \"\"\"\n    valid_indices = []\n    \n    for i, seg in enumerate(segments):\n        # Check if any channel exceeds threshold\n        if np.max(np.abs(seg)) < print(f\"Remaining: {len(clean_segments)} segments\")\n    \n    return clean_segments, valid_indices\n\n# Usage\nclean_segments, valid_idx = remove_artifact_segments(segments,
